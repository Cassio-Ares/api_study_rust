# Routes


````
// web -> contém as ferramentas de roteamento
use actix_web::web;
// Importa os handlers que serão associados a cada rota
use crate::handlers::payment_handlers;



//"cfg: &mut web::ServiceConfig" -> recebe o configurador do Actix por referência mutável
pub fn config(cfg: &mut web::ServiceConfig) {
  cfg.service(
    // web::resource -> aninha as rotas e todas dentro dela começam com "/payment"
    web::resource("/payment")

   // route-> rota 
   // web::verbo(get, post, put, ...) -> define a função da rota
   // to -> carrega o handler que será usado 
    .route(web::post().to(payment_handlers::create_payment_handler))

    .route(web::get().to(payment_handlers::get_all_payments_handler))

    // .service -> sub-recurso dentro de web::resource("/payment")
    // todas as rotas aqui serão /payment/{uuid}
    .service(
       web::resource("/{uuid}")

        .route(web::get().to(payment_handlers::get_payment_by_uuid_handler))

        .route(web::put().to(payment_handlers::update_payment_handler))

        .route(web::patch().to(payment_handlers::update_payment_status_handler))

        // .service dentro do service ->  sub-recurso: /payment/{uuid}/refund
        //aninhado dentro de /{uuid} — consistente com a hierarquia
        .service(
           web::resource("/refund")

            .route(web::patch().to(payment_handlers::refund_payment_handler))
        )
    )
  )
}
````
